Q. create a simple custom hook example 
A. 1. create custom hook useCounter.js
  2. import hook in component and use it

  1. useCounter.js
      import { useState } from "react";
      
      function useCounter() {
        const [count, setCount] = useState(0);
      
        const increment = () => setCount(c => c + 1);
        const decrement = () => setCount(c => c - 1);
      
        return { count, increment, decrement };
      }      
      export default useCounter;

    2. 
      import useCounter from "./useCounter";
      
      function App() {
        const { count, increment, decrement } = useCounter();
      
        return (
          <div>
            <h2>Count: {count}</h2>
      
            <button onClick={increment}>+</button>
            <button onClick={decrement}>-</button>
          </div>
        );
      }
      export default App;
Q. call a api and bind its data in html, add a search box, if no data then show "No Data"
A. 
  function App() {
    const [userData,setUserData]=useState([]);
    useEffect(()=>{
    fetch("https://jsonplaceholder.typicode.com/users")

Q. show small example to share data using context
A. 1. create context file 
    2. provide data, here sharing from app component
    3. consume data in child component

    1. UserContext.jsx
      import { createContext } from "react";
      export const UserContext = createContext();

    2. App.jsx
        import { UserContext } from "./UserContext";
        import Child from "./Child";
        
        function App() {
          const user = {
            name: "Gautam",
            role: "Developer"
          };

          return (
            <UserContext.Provider value={user}>
              <h2>Parent Component</h2>
              <Child />
            </UserContext.Provider>
          );
        }
        export default App;

      3. Child.jsx
        import { useContext } from "react";
        import { UserContext } from "./UserContext";
        
        function Child() {
          const user = useContext(UserContext);

          return (
            <div>
              <h3>Child Component</h3>
              <p>Name: {user.name}</p>
              <p>Role: {user.role}</p>
            </div>
          );
        }        
        export default Child;
